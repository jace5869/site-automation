pywinrm[kerberos]
requests-kerberos
requests-ntlm          # NTLM: not on a FIPS host (no MD4); Kerberos is the way there
pypsrp[kerberos]       # the psrp connection, an alternative to winrm
# ---- Active Directory over LDAP, from the EE (microsoft.ad brings sansldap, pyspnego, dnspython) ----
ldap3                  # pure Python: your own LDAP scripts
# ---- SQL Server (community.general.mssql_db, mssql_script) - ansible.mysql brings PyMySQL itself ----
pymssql                # built from its source, with the EE's own OpenSSL: PyPI's ready-made one stops Python on a
                       # FIPS host (execution-environment.yml, prepend_builder; freetds.tar.gz)
mssql-python           # Microsoft's SQL Server driver, for your own Python (import mssql_python; the mssql_* modules
                       # use pymssql): its ODBC Driver 18 is in the wheel, Kerberos through the system's krb5
# ---- general ----
jmespath               # the json_query filter
dnspython              # the community.general.dig lookup
smbprotocol            # the report website on an SMB share (report_web_dest \\server\share): Kerberos through pyspnego
